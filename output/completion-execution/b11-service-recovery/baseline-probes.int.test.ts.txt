let t: TestDatabase;
beforeEach(async () => { t = await createTestDatabase(); probe.db = t.db; });

const future = () => new Date(Date.now() + 3600000).toISOString();
const day = (offset: number) => new Date(Date.now() + offset * 86400000).toISOString().slice(0,10);
const context = () => ({ ip: "127.0.0.1", receiptSession: newReceiptSession() });
const contact = () => ({ kind: "email", value: `synthetic-${randomUUID()}@example.test` });
async function receiveAndAccept(f: Awaited<ReturnType<typeof syntheticServiceFixture>>, payload: Record<string, unknown>) {
  const submitted = await submitInquiry(t.db, payload, context());
  const [inquiry] = await t.db.select().from(inquiries).where(eq(inquiries.reference, submitted.receipt.reference));
  const accepted = await acceptInquiry(t.db, f.session, { id: inquiry!.id, expectedVersion: inquiry!.version,
    operationId: randomUUID(), nextAction: "Synthetic human clarification", dueAt: future() });
  return { inquiry: inquiry!, accepted };
}
it("B11 recovery: a past stay start can be corrected using the same submission key", async () => {
  const f = await syntheticServiceFixture(t.db);
  const offered = f.document.decisions.find(row => row.state === "offered");
  if (offered?.state !== "offered") throw new Error("Missing offering");
  const scope = await publishSyntheticScope(t.db, f.session, {
    kind: "bounded_service_scope_v1", decisions: [{ ...offered, topic: "short_stay_consultation" }]
  });
  const payload = { submissionKey: issueSubmissionKey(), locale: "bg", purpose: "service_consultation",
    contact: contact(), privacyNotice: true, contentReference: { kind: f.source.kind, slug: f.source.slug, versionId: f.source.versionId },
    serviceRequest: { ...f.request, topic: "short_stay_consultation", scopeVersionId: scope.outcome.id,
      relationship: undefined, requestedFrom: day(-1), requestedUntil: day(2) } };
  const ctx = context();
  await expect(submitInquiry(t.db, payload, ctx)).rejects.toMatchObject({ code: "validation_failed", fieldErrors: { serviceRequest: ["future_required"] } });
  const receipt = await t.db.select({status: operations.status, outcome: operations.outcome}).from(operations).where(eq(operations.idempotencyKey, payload.submissionKey));
  console.log("B11 past-date retained operation", JSON.stringify(receipt));
  await expect(submitInquiry(t.db, { ...payload, serviceRequest: { ...payload.serviceRequest, requestedFrom: day(1), requestedUntil: day(3) } }, ctx)).resolves.toMatchObject({ receipt: { status: "accepted" } });
});
it("B11 service boundary: a brokerage inquiry cannot link to a service Case without service context", async () => {
  const f = await syntheticServiceFixture(t.db), route = contact();
  const a = await receiveAndAccept(f, { submissionKey: issueSubmissionKey(), locale: "bg", purpose: "service_consultation",
    contact: route, privacyNotice: true, contentReference: { kind: f.source.kind, slug: f.source.slug, versionId: f.source.versionId }, serviceRequest: f.request });
  const created = await createCaseFromInquiry(t.db, f.session, { id: a.inquiry.id, operationId: randomUUID(),
    expectedVersion: a.accepted.outcome.version, kind: "service_intake", serviceTopic: f.request.topic,
    title: "Synthetic service Case", nextAction: "Synthetic clarification", dueAt: future(), requirements: "Synthetic service need", preferences: "" });
  const b = await receiveAndAccept(f, { submissionKey: issueSubmissionKey(), locale: "bg", purpose: "seller_consultation",
    contact: route, privacyNotice: true, message: "Synthetic seller brokerage request" });
  const candidates = await listInquiryCaseCandidates(t.db, f.session, b.inquiry.id);
  console.log("B11 nonservice candidate", JSON.stringify(candidates.map(row => ({kind:row.kind, canLink:row.canLink, blockReason:row.blockReason}))));
  await expect(linkInquiryToExistingCase(t.db, f.session, { id: b.inquiry.id, operationId: randomUUID(), expectedVersion: b.accepted.outcome.version,
    caseId: created.outcome.id, expectedCaseVersion: created.outcome.version })).rejects.toMatchObject({ code: "transition_denied" });
});
it("B11 scope form: identical save replay returns its receipt after an unrelated draft change", async () => {
  const f = await syntheticServiceFixture(t.db); probe.session = f.session;
  const state = initialFormState("service_scope.save", { topic: "short_stay_consultation", region: JSON.stringify({id:f.region.id,version:f.region.version}), state:"unknown" }, (await readServiceScope(t.db, f.session)).draft.version);
  const data = new FormData();
  data.set(formFields.intent, "submit"); data.set(formFields.operationId, state.operationId); data.set(formFields.expectedRevision, String(state.expectedRevision));
  for (const [key,value] of Object.entries(state.values)) data.set(key,value);
  const first = await submitServiceScope("bg", "save", state, data);
  expect(first.outcome.kind).toBe("confirmed");
  const draft = (await readServiceScope(t.db, f.session)).draft;
  const changed = structuredClone(draft.document);
  const offering = changed.decisions.find(row => row.state === "offered");
  if (offering?.state !== "offered") throw new Error("Missing offering");
  offering.mode = "consultation";
  await saveServiceScope(t.db, f.session, { operationId: randomUUID(), expectedVersion: draft.version, document: changed });
  const replay = await submitServiceScope("bg", "save", state, data);
  console.log("B11 scope form replay", JSON.stringify({ first: first.outcome, replay: replay.outcome }));
  expect(replay.outcome.kind).toBe("confirmed");
});

it("B11 locale continuation: revoked observed locale holds new consultation", async () => {
  const { readContentLocaleWorkbench, saveContentLocale, decideContentLocale } = await import("@/server/content/locale-commands");
  const { readApprovedContent } = await import("@/server/content/public");
  const { readCase } = await import("@/server/cases/queries");
  const { transitionCaseStage } = await import("@/server/cases/lifecycle");
  const f = await syntheticServiceFixture(t.db);
  await t.db.insert(grants).values({ principalId:f.actor.id, capability:"translation.review", reason:"Synthetic locale probe" });
  const bound = async () => {
    const view = await readContentLocaleWorkbench(t.db, f.session, f.contentId, "en");
    if (!view.source) throw new Error("Missing approved synthetic source");
    return { id:f.contentId, locale:"en" as const, operationId:randomUUID(), expectedVersion:view.edition?.version ?? 1,
      sourceExpectedVersion:view.page.version, sourceVersionId:view.source.version.id, sourceVersionHash:view.source.version.contentHash };
  };
  await saveContentLocale(t.db, f.session, { ...(await bound()), title:"Synthetic EN service", text:"Synthetic EN consultation terms only" });
  for (const decision of ["request_review","claims","language","editorial","publish"] as const)
    await decideContentLocale(t.db, f.session, { ...(await bound()), decision, reviewed:true,
      note:"Synthetic exact locale decision", expiresAt:new Date(Date.now()+86400000).toISOString() });
  const localized = await readApprovedContent(t.db, "service", f.source.slug, "en");
  expect(localized).not.toBeNull();
  const a = await receiveAndAccept(f, { submissionKey:issueSubmissionKey(), locale:"en", purpose:"service_consultation",
    contact:contact(), privacyNotice:true, contentReference:{kind:"service",slug:f.source.slug,versionId:localized!.version.id}, serviceRequest:f.request });
  const created = await createCaseFromInquiry(t.db, f.session, { id:a.inquiry.id, operationId:randomUUID(), expectedVersion:a.accepted.outcome.version,
    kind:"service_intake", serviceTopic:f.request.topic, title:"Synthetic EN consultation", nextAction:"Synthetic clarification", dueAt:future(), requirements:"Synthetic service need", preferences:"" });
  await decideContentLocale(t.db, f.session, { ...(await bound()), decision:"withdraw", reviewed:true, note:"Synthetic human locale withdrawal" });
  expect(await readApprovedContent(t.db, "service", f.source.slug, "en")).toBeNull();
  const view = await readCase(t.db, f.session, created.outcome.id);
  const result = await transitionCaseStage(t.db, f.session, { id:created.outcome.id, expectedVersion:created.outcome.version,
    operationId:randomUUID(), stage:"consultation", reason:"Synthetic new consultation after locale withdrawal", outcome:"", completionEvidenceId:"", retention:"", aftercare:"", handover:"" });
  console.log("B11 revoked locale continuation", JSON.stringify({ current:view.record.service?.current, transition:result.outcome }));
  expect(view.record.service?.current).toBe(false);
});
